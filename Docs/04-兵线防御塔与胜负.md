## 1. 先建立兵线配置

创建 `Core/LaneConfig.cs`。下面是可直接使用的**完整文件**；读写方法预先为阶段 06/09 的配置序列化准备。


```csharp
// 完整文件：Assets/Moba/Runtime/Core/LaneConfig.cs
using System;
using System.IO;

namespace Moba.Core
{
    public sealed class LaneConfig
    {
        public readonly int MinionHp, MinionSpeed, MinionDamage, MinionRange, MinionWindup, MinionCooldown, AcquireRadius;
        public readonly int TowerDamage, TowerRange, TowerWindup, TowerCooldown;
        public readonly int WaveInterval, WaveSize, MinionCapPerTeam, RecoveryRadius, RecoveryPerFrame, RecoveryDelay;
        public Int2 Midpoint => default;

        public LaneConfig(int minionHp = 180, int minionSpeed = 150, int minionDamage = 15, int minionRange = 1200,
            int minionWindup = 6, int minionCooldown = 24, int acquireRadius = 6000,
            int towerDamage = 100, int towerRange = 8000, int towerWindup = 4, int towerCooldown = 20,
            int waveInterval = 300, int waveSize = 3, int minionCapPerTeam = 24,
            int recoveryRadius = 3000, int recoveryPerFrame = 2, int recoveryDelay = 60)
        {
            if (minionHp < 1 || minionHp > 100000 || minionSpeed < 2 || minionSpeed > 1000 ||
                minionDamage < 1 || minionDamage > 100000 || minionRange < 1 || minionRange > 80000 ||
                minionWindup < 1 || minionWindup > 1200 || minionCooldown <= minionWindup || minionCooldown > 12000 ||
                acquireRadius < minionRange || acquireRadius > 80000 || towerDamage < 1 || towerDamage > 100000 ||
                towerRange < 1 || towerRange > 80000 || towerWindup < 1 || towerWindup > 1200 ||
                towerCooldown <= towerWindup || towerCooldown > 12000 || waveInterval < 1 || waveInterval > 12000 ||
                waveSize < 1 || waveSize > 24 || minionCapPerTeam < waveSize || minionCapPerTeam > 24 ||
                recoveryRadius < 1 || recoveryRadius > 80000 || recoveryPerFrame < 1 || recoveryPerFrame > 100000 ||
                recoveryDelay < 1 || recoveryDelay > 12000) throw new ArgumentException("Invalid lane configuration.");
            MinionHp = minionHp; MinionSpeed = minionSpeed; MinionDamage = minionDamage; MinionRange = minionRange;
            MinionWindup = minionWindup; MinionCooldown = minionCooldown; AcquireRadius = acquireRadius;
            TowerDamage = towerDamage; TowerRange = towerRange; TowerWindup = towerWindup; TowerCooldown = towerCooldown;
            WaveInterval = waveInterval; WaveSize = waveSize; MinionCapPerTeam = minionCapPerTeam;
            RecoveryRadius = recoveryRadius; RecoveryPerFrame = recoveryPerFrame; RecoveryDelay = recoveryDelay;
        }

        internal void Write(BinaryWriter w)
        {
            w.Write(MinionHp); w.Write(MinionSpeed); w.Write(MinionDamage); w.Write(MinionRange);
            w.Write(MinionWindup); w.Write(MinionCooldown); w.Write(AcquireRadius);
            w.Write(TowerDamage); w.Write(TowerRange); w.Write(TowerWindup); w.Write(TowerCooldown);
            w.Write(WaveInterval); w.Write(WaveSize); w.Write(MinionCapPerTeam);
            w.Write(RecoveryRadius); w.Write(RecoveryPerFrame); w.Write(RecoveryDelay);
        }

        internal static LaneConfig Read(BinaryReader r) => new LaneConfig(
            r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(),
            r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(), r.ReadInt32(),
            r.ReadInt32(), r.ReadInt32(), r.ReadInt32());
    }
}
```


每 300 帧一波意味着 15 秒；NextWaveFrame 初值为 1，所以开局第 1 帧就会出第一波。每队上限是活着的小兵数量，不是历史创建总数。

## 3. 实现稳定选敌

创建 `TargetSelector.cs`：


```csharp
// 完整文件：Assets/Moba/Runtime/Core/TargetSelector.cs
namespace Moba.Core
{
    public enum TargetPriority { Robot, Minion, Tower }

    public static class TargetSelector
    {
        public static int Rank(EntityState target, TargetPriority priority)
        {
            if (priority == TargetPriority.Tower) return target.Kind == EntityKind.Minion ? 0 : target.IsHero ? 1 : 2;
            if (priority == TargetPriority.Robot) return target.IsHero ? 0 : target.Kind == EntityKind.Minion ? 1 : 2;
            return target.Kind == EntityKind.Tower ? 1 : 0;
        }

        public static EntityState Find(CombatWorld world, EntityState source, int radius, TargetPriority priority)
        {
            EntityState best = null; int rank = int.MaxValue; long distance = long.MaxValue;
            if (source == null || !source.Alive) return null;
            foreach (var candidate in world.State.Entities)
            {
                if (!candidate.Alive || candidate.Team == source.Team ||
                    (priority == TargetPriority.Tower && candidate.Kind == EntityKind.Tower)) continue;
                long d = IntegerMath.DistanceSquared(source.Position, candidate.Position);
                if (d > (long)radius * radius) continue;
                int r = Rank(candidate, priority);
                if (r > rank || (r == rank && (d > distance || (d == distance && candidate.EntityId > best.EntityId)))) continue;
                best = candidate; rank = r; distance = d;
            }
            return best;
        }
    }
}
```


选择顺序是“类型优先级 → 距离平方 → EntityId”，不是依赖 Physics 返回顺序。塔优先小兵，其次英雄；机器人优先英雄；小兵把塔放在更低优先级。

在测试中摆两名等距同类敌人，确认始终选 EntityId 较小者。再把集合顺序反转，选择不应变化。

## 5. 实现 LaneRules

新建以下**完整文件** `Core/LaneRules.cs`。这是去除后期 SpawnPlacement 依赖的教学版，允许初始单位重叠；阶段 12 再替换出生分配。


```csharp
using System;

namespace Moba.Core
{
    public sealed partial class CombatWorld
    {
        public Int2 SpawnFor(Team team) => team == Team.Blue ? Config.BlueSpawn : Config.RedSpawn;
        public Int2 EnemyTowerFor(Team team) => team == Team.Blue ? Config.RedTower : Config.BlueTower;
        public int MaxHp(EntityState entity) => entity.Kind == EntityKind.Tower ? Config.TowerHp : entity.Kind == EntityKind.Minion ? Config.Lane.MinionHp : Config.HeroHp;

        private void BeginMatchFrame()
        {
            foreach (var entity in entities)
            {
                if (!entity.IsHero || entity.Alive || entity.RespawnFrame == 0 || CurrentFrame < entity.RespawnFrame) continue;
                var respawn = SpawnFor(entity.Team);
                entity.Hp = Config.HeroHp; entity.Position = respawn;
                entity.Intent = CombatCommand.NoOp; entity.HitFrame = 0; entity.RespawnFrame = 0;
                entity.AttackReadyFrame = 0; entity.SkillReadyFrame = 0; entity.LastDamagedFrame = -1; entity.PathIndex = 0;
                CombatFsm.Transition(this, entity, FsmState.Idle);
                Emit(new SimEvent(CurrentFrame, SimEventKind.Respawned, entity.EntityId));
            }
            if (CurrentFrame != State.NextWaveFrame) return;
            for (int teamIndex = 0; teamIndex < 2; teamIndex++)
            {
                var team = (Team)teamIndex; int living = 0;
                foreach (var entity in entities) if (entity.Kind == EntityKind.Minion && entity.Team == team && entity.Alive) living++;
                int count = Math.Min(Config.Lane.WaveSize, Config.Lane.MinionCapPerTeam - living);
                for (int i = 0; i < count; i++)
                {
                    var spawn = new Int2(SpawnFor(team).X, (i - 1) * 1000);
                    var entity = new EntityState { EntityId = State.NextEntityId++, Kind = EntityKind.Minion,
                        Team = team, Position = spawn, Hp = Config.Lane.MinionHp };
                    entities.Add(entity); Emit(new SimEvent(CurrentFrame, SimEventKind.Spawned, entity.EntityId));
                }
            }
            State.WaveNumber++; State.NextWaveFrame = checked(CurrentFrame + Config.Lane.WaveInterval);
        }

        private void UpdateNpcIntents()
        {
            foreach (var entity in entities)
            {
                if (!entity.Alive || entity.IsHero) continue;
                bool tower = entity.Kind == EntityKind.Tower;
                int radius = tower ? Config.Lane.TowerRange : Config.Lane.AcquireRadius;
                var priority = tower ? TargetPriority.Tower : TargetPriority.Minion;
                var target = TargetSelector.Find(this, entity, radius, priority);
                var current = entity.Intent.Kind == CommandKind.Attack ? FindEntity(entity.Intent.TargetId) : null;
                // Keep a valid target within its priority class, avoiding perpetual windup restarts as distances change.
                if (IsEnemy(entity, current) && IntegerMath.WithinRange(entity.Position, current.Position, radius) &&
                    (target == null || TargetSelector.Rank(current, priority) <= TargetSelector.Rank(target, priority))) target = current;
                if (target != null) Apply(entity, new CombatCommand(CommandKind.Attack, default, target.EntityId));
                else if (tower) ClearIntent(entity);
                else
                {
                    if (entity.PathIndex == 0 && (IntegerMath.WithinRange(entity.Position, Config.Lane.Midpoint, 1500) || (entity.Team == Team.Blue ? entity.Position.X >= 0 : entity.Position.X <= 0))) entity.PathIndex = 1;
                    Int2 destination = entity.PathIndex == 0 ? Config.Lane.Midpoint : EnemyTowerFor(entity.Team);
                    if (entity.Intent.Kind != CommandKind.MoveTo || entity.Intent.Point != destination)
                        Apply(entity, CombatCommand.MoveTo(destination));
                }
            }
        }

        private void EndMatchFrame()
        {
            foreach (var entity in entities)
            {
                if (!entity.IsHero || !entity.Alive || entity.Hp >= Config.HeroHp ||
                    !IntegerMath.WithinRange(entity.Position, SpawnFor(entity.Team), Config.Lane.RecoveryRadius) ||
                    (entity.LastDamagedFrame >= 0 && CurrentFrame - entity.LastDamagedFrame < Config.Lane.RecoveryDelay)) continue;
                int amount = Math.Min(Config.Lane.RecoveryPerFrame, Config.HeroHp - entity.Hp);
                entity.Hp += amount; Emit(new SimEvent(CurrentFrame, SimEventKind.Healed, entity.EntityId, entity.EntityId, amount));
            }
            entities.RemoveAll(e => e.Kind == EntityKind.Minion && !e.Alive);
            bool blueDead = !FindEntity(5).Alive, redDead = !FindEntity(6).Alive;
            if (blueDead && redDead) State.Result = MatchResult.Draw;
            else if (blueDead) State.Result = MatchResult.RedWins;
            else if (redDead) State.Result = MatchResult.BlueWins;
        }
    }
}
```


按照三个函数理解：

- BeginMatchFrame：先复活，再按固定队伍顺序生成一波小兵，更新 NextEntityId、WaveNumber、NextWaveFrame。
- UpdateNpcIntents：只处理小兵与塔；已有目标在同优先级内仍有效时保留，避免持续重启前摇。
- EndMatchFrame：回复 → 移除死亡小兵 → 根据两座塔生命裁定胜负。

恢复区用整数距离；LastDamagedFrame 为 -1 代表从未受伤。不要把 `Time.time` 写入 Core。

## 6. 接到 Step 的正确位置

在现有 CombatWorld.Step 中改三处，其他顺序不变：

1. 原 `BeginCombatFrame();` 改为：
```csharp
if (Config.Mode == SimulationMode.Match) BeginMatchFrame();
else BeginCombatFrame();
```

2. 玩家和机器人 Apply 完成后、Movement 阶段前插入：
```csharp
if (Config.Mode == SimulationMode.Match) UpdateNpcIntents();
```

3. ResolveDamage 后、State.Frame 赋值前插入：
```csharp
if (Config.Mode == SimulationMode.Match) EndMatchFrame();
```

两个 Begin 方法不会在一帧同时执行。保留 BeginCombatFrame 给纯战斗测试使用；Match 使用有兵线的 BeginMatchFrame。

ManualLocalBootstrap 已在 World.Result 不为 None 时停止推进，不要删除这个判断。玩家输入在结算后也应 CanControl=false 并 ClearPending。

先写 EditMode 测试，使用 `new BattleConfig(mode: SimulationMode.Match, lane: new LaneConfig(waveInterval: 10))` 缩短等待。场景正式配置仍使用 300 帧。

| 测试 | 执行方式 | 应断言 |
| --- | --- | --- |
| 开局波次 | 执行帧 1 | 每队 3 个小兵，NextWaveFrame=301（默认配置） |

## 8. 让场景对局可观察

在视图中展示蓝红塔 HP、当前波次和胜负文本。实体根部可增加血条；通过 World.MaxHp(entity) 计算比例，不让所有类型都除以 HeroHp。

右键敌塔应先走入攻击范围再攻击。小兵相遇后应停下交战，而不是在每帧选敌时重置 HitFrame。英雄回血先观察数字，再添加出生区特效。

如果默认局需要较长时间，可以在测试中降低塔 HP验证结算，但最终必须用记录清楚的正式配置完成一次自然胜负，不把限帧退出当作摧塔。

## 9. 常见问题

| 现象 | 排查 |
| --- | --- |
| 没有小兵 | Mode 是否为 Match；是否调用 BeginMatchFrame |
| 小兵一直绕在中点 | PathIndex 是否在到达或越过中点时变为 1 |
| 塔追着玩家跑 | MoveSpeed(Tower) 必须为 0 |
| 塔攻击前摇反复重置 | 是否保留有效目标，并避免重复 Apply 同一攻击 |
| 小兵死后显示仍在 | BattleView.Refresh 是否清理世界中已移除的 ID |
| 回复从不发生 | LastDamagedFrame 默认值、范围单位、帧差边界 |
| 编译报重复 SpawnFor | 同一 partial 类只能保留一个同名方法实现 |
